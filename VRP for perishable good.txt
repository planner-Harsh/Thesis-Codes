def load_roads(shp_path, default_speed=50):
    gdf = gpd.read_file(shp_path)
    print(f"Loaded {len(gdf)} features")
    if gdf.crs is None or gdf.crs.is_geographic:
        gdf = gdf.to_crs("EPSG:32643")
    # Explode MultiLineString
    rows = []
    for idx, row in gdf.iterrows():
        geom = row.geometry
        if geom is None:
            continue
        if geom.geom_type == 'MultiLineString':
            for part in geom.geoms:
                new_row = row.copy()
                new_row.geometry = part
                rows.append(new_row)
        else:
            rows.append(row)
    gdf = gpd.GeoDataFrame(rows, crs=gdf.crs)
    gdf['length_m'] = gdf.geometry.length
    if 'speed_kmh' in gdf.columns:
        gdf['speed_kmh'] = gdf['speed_kmh'].fillna(default_speed)
    else:
        gdf['speed_kmh'] = default_speed
    # Environmental defaults
    for col, default in [('fl_depth_m', 0.0), ('heat_lst_c', 35.0), ('can_shade', 0.0)]:
        if col not in gdf.columns:
            gdf[col] = default
    return gdf

# 3. Compute costs (if columns are constant, cost will be constant)
# ------------------------------------------------------
lst_min, lst_max = gdf['heat_lst_c'].min(), gdf['heat_lst_c'].max()
can_min, can_max = gdf['can_shade'].min(), gdf['can_shade'].max()
def normalize(x, mn, mx):
    return 0.5 if mx == mn else (x - mn) / (mx - mn)
gdf['lst_std'] = gdf['heat_lst_c'].apply(lambda x: normalize(x, lst_min, lst_max))
gdf['canopy_std'] = gdf['can_shade'].apply(lambda x: normalize(x, can_min, can_max))
gdf['canopy_pen'] = 1 - gdf['canopy_std']
gdf['cost_summer'] = 0.65 * gdf['lst_std'] + 0.35 * gdf['canopy_pen']
gdf['time_base_sec'] = gdf['length_m'] / (gdf['speed_kmh'] * 1000 / 3600)
gdf['time_summer_sec'] = gdf['time_base_sec'] * gdf['cost_summer']

def monsoon_time(row):
    L, Vn, d = row['length_m'], row['speed_kmh'], row['fl_depth_m']
    if d < 0.10:
        return L / (Vn * 1000 / 3600)
    elif 0.10 <= d <= 0.46:
        factor = 1 - ((d - 0.10) / 0.36) * 0.90
        Vm = Vn * factor
        return L / (Vm * 1000 / 3600) if Vm > 0 else 999999
    else:
        return 999999
gdf['time_monsoon_sec'] = gdf.apply(monsoon_time, axis=1)

print(f"Summer cost range: {gdf['cost_summer'].min():.3f} – {gdf['cost_summer'].max():.3f}")

print("Building graph...")
G = nx.MultiDiGraph()
for idx, row in gdf.iterrows():
    geom = row.geometry
    if geom.geom_type != 'LineString':
        continue
    coords = list(geom.coords)
    for i in range(len(coords)-1):
        u, v = coords[i], coords[i+1]
        if u not in G.nodes:
            G.add_node(u, x=u[0], y=u[1])
        if v not in G.nodes:
            G.add_node(v, x=v[0], y=v[1])
        sub_len = row['length_m'] / (len(coords)-1)
        t_summer = row['time_summer_sec'] / (len(coords)-1)
        t_monsoon = row['time_monsoon_sec'] / (len(coords)-1)
        G.add_edge(u, v, length=sub_len, time_summer=t_summer, time_monsoon=t_monsoon)
        G.add_edge(v, u, length=sub_len, time_summer=t_summer, time_monsoon=t_monsoon)
print(f"Graph nodes: {G.number_of_nodes()}, edges: {G.number_of_edges()}")

# 6. Helper: nearest graph node
# ------------------------------------------------------
def nearest_node(G, lon, lat):
    point = Point(lon, lat)
    point_utm = gpd.GeoSeries([point], crs="EPSG:4326").to_crs("EPSG:32643").iloc[0]
    best_node, best_dist = None, float('inf')
    for node, data in G.nodes(data=True):
        dx = data['x'] - point_utm.x
        dy = data['y'] - point_utm.y
        dist = dx*dx + dy*dy
        if dist < best_dist:
            best_dist = dist
            best_node = node
    return best_node

# 7. Function to solve VRP and collect geometries safely
# ------------------------------------------------------
def solve_scenario(scenario_name, time_weight):
    print(f"\n{'='*60}")
    print(f"RUNNING {scenario_name.upper()} SCENARIO")
    print('='*60)
    start_time = time.time()

    clusters = {
        "CC-P1": villages[villages['cc'] == 'CC-P1'],
        "CC-S1": villages[villages['cc'] == 'CC-S1'],
        "CC-S2": villages[villages['cc'] == 'CC-S2'],
        "CC-E1": villages[villages['cc'] == 'CC-E1'],
    }

    all_routes = {}
    first_level_trips = []
    route_geometries = []

    # First level: villages → CC
    for cc_name, vills in clusters.items():
        if len(vills) == 0:
            continue
        print(f"\nProcessing {cc_name} with {len(vills)} villages...")
        cc_row = ccs[ccs['name'] == cc_name].iloc[0]
        cc_node = nearest_node(G, cc_row['lon'], cc_row['lat'])
        if cc_node is None:
            print(f"  Cannot map CC {cc_name} to graph")
            continue
        v_nodes, supplies, names = [], [], []
        for _, v in vills.iterrows():
            node = nearest_node(G, v['lon'], v['lat'])
            if node is None:
                print(f"  Warning: could not map {v['name']} to graph")
                continue
            v_nodes.append(node)
            supplies.append(v['supply'])
            names.append(v['name'])
        if len(v_nodes) == 0:
            continue
        nodes = [cc_node] + v_nodes
        n = len(nodes)
        time_matrix = np.zeros((n, n), dtype=int)
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                try:
                    t = nx.shortest_path_length(G, nodes[i], nodes[j], weight=time_weight)
                    time_matrix[i][j] = int(t)
                except nx.NetworkXNoPath:
                    time_matrix[i][j] = 86400
        demands = [0] + supplies
        total_demand = sum(supplies)
        capacity_per_truck = 15000
        num_trucks = max(1, int(math.ceil(total_demand / capacity_per_truck)))
        vehicle_capacities = [capacity_per_truck] * num_trucks
        print(f"  {cc_name}: total demand {total_demand} MT → {num_trucks} trucks of {capacity_per_truck} MT")
        data = {
            'distance_matrix': time_matrix.tolist(),
            'num_vehicles': num_trucks,
            'depot': 0,
            'demands': demands,
            'vehicle_capacities': vehicle_capacities
        }
        manager = pywrapcp.RoutingIndexManager(n, data['num_vehicles'], data['depot'])
        routing = pywrapcp.RoutingModel(manager)
        def distance_callback(from_idx, to_idx):
            return data['distance_matrix'][manager.IndexToNode(from_idx)][manager.IndexToNode(to_idx)]
        transit_cb = routing.RegisterTransitCallback(distance_callback)
        routing.SetArcCostEvaluatorOfAllVehicles(transit_cb)
        def demand_callback(from_idx):
            return data['demands'][manager.IndexToNode(from_idx)]
        demand_cb = routing.RegisterUnaryTransitCallback(demand_callback)
        routing.AddDimensionWithVehicleCapacity(demand_cb, 0, data['vehicle_capacities'], True, 'Capacity')
        params = pywrapcp.DefaultRoutingSearchParameters()
        params.first_solution_strategy = routing_enums_pb2.FirstSolutionStrategy.PATH_CHEAPEST_ARC
        solution = routing.SolveWithParameters(params)
        if solution:
            routes = []
            for v in range(data['num_vehicles']):
                idx = routing.Start(v)
                route_nodes = []
                while not routing.IsEnd(idx):
                    route_nodes.append(manager.IndexToNode(idx))
                    idx = solution.Value(routing.NextVar(idx))
                routes.append(route_nodes)
            all_routes[cc_name] = (routes, nodes, names)
            for truck_idx, route_nodes in enumerate(routes):
                # Build geometry for this truck's closed tour
                all_coords = []
                for i in range(len(route_nodes)-1):
                    u, v = nodes[route_nodes[i]], nodes[route_nodes[i+1]]
                    try:
                        path = nx.shortest_path(G, u, v, weight=time_weight)
                        for node in path:
                            all_coords.append((G.nodes[node]['x'], G.nodes[node]['y']))
                    except:
                        pass
                # Return to depot
                last_node = nodes[route_nodes[-1]]
                first_node = nodes[route_nodes[0]]
                try:
                    path = nx.shortest_path(G, last_node, first_node, weight=time_weight)
                    for node in path[1:]:
                        all_coords.append((G.nodes[node]['x'], G.nodes[node]['y']))
                except:
                    pass
                # Remove consecutive duplicates
                unique_coords = []
                for p in all_coords:
                    if not unique_coords or (abs(p[0]-unique_coords[-1][0])>1e-9 or abs(p[1]-unique_coords[-1][1])>1e-9):
                        unique_coords.append(p)
                if len(unique_coords) >= 2:
                    geom = LineString(unique_coords)
                    route_geometries.append({
                        'geometry': geom,
                        'scenario': scenario_name,
                        'level': 'village_to_CC',
                        'cc': cc_name,
                        'truck_id': truck_idx
                    })
                # Compute total distance for metrics
                total_dist = 0
                for i in range(len(route_nodes)-1):
                    u, v = nodes[route_nodes[i]], nodes[route_nodes[i+1]]
                    try:
                        path_len = nx.shortest_path_length(G, u, v, weight=time_weight)
                        dist_km = path_len / 3600 * 50
                        total_dist += dist_km
                    except:
                        pass
                last_node = nodes[route_nodes[-1]]
                first_node = nodes[route_nodes[0]]
                try:
                    path_len = nx.shortest_path_length(G, last_node, first_node, weight=time_weight)
                    dist_km = path_len / 3600 * 50
                    total_dist += dist_km
                except:
                    pass
                first_level_trips.append({
                    'cc': cc_name,
                    'truck_id': truck_idx,
                    'distance_km': total_dist
                })
        else:
            print(f"  ❌ No solution for {cc_name}")

    # Second level: CC → APMC
    second_level_trips = []
    for cc_name, apmc_name in cc_to_apmc.items():
        if cc_name not in all_routes:
            continue
        cc_row = ccs[ccs['name'] == cc_name].iloc[0]
        apmc_row = apmcs[apmcs['name'] == apmc_name].iloc[0]
        cc_node = nearest_node(G, cc_row['lon'], cc_row['lat'])
        apmc_node = nearest_node(G, apmc_row['lon'], apmc_row['lat'])
        if cc_node is None or apmc_node is None:
            continue
        try:
            path = nx.shortest_path(G, cc_node, apmc_node, weight=time_weight)
            coords = [(G.nodes[node]['x'], G.nodes[node]['y']) for node in path]
            if len(coords) >= 2:
                geom = LineString(coords)
                route_geometries.append({
                    'geometry': geom,
                    'scenario': scenario_name,
                    'level': 'CC_to_APMC',
                    'cc': cc_name,
                    'apmc': apmc_name
                })
            t = nx.shortest_path_length(G, cc_node, apmc_node, weight=time_weight)
            dist_km = t / 3600 * 50
            second_level_trips.append({
                'cc': cc_name,
                'apmc': apmc_name,
                'distance_km': dist_km
            })
            print(f"  {cc_name} → {apmc_name}: {t/60:.1f} minutes ({dist_km:.1f} km)")
        except nx.NetworkXNoPath:
            print(f"  ⚠️ No path between {cc_name} and {apmc_name}")

    # Direct villages to Padra APMC
    direct_trips = []
    direct_vills = villages[villages['cc'] == 'DIRECT']
    padra_node = nearest_node(G, apmcs[apmcs['name']=='Padra_APMC'].iloc[0]['lon'],
                              apmcs[apmcs['name']=='Padra_APMC'].iloc[0]['lat'])
    for _, v in direct_vills.iterrows():
        v_node = nearest_node(G, v['lon'], v['lat'])
        if v_node is None:
            continue
        try:
            path = nx.shortest_path(G, v_node, padra_node, weight=time_weight)
            coords = [(G.nodes[node]['x'], G.nodes[node]['y']) for node in path]
            if len(coords) >= 2:
                geom = LineString(coords)
                route_geometries.append({
                    'geometry': geom,
                    'scenario': scenario_name,
                    'level': 'direct_village_to_APMC',
                    'village': v['name'],
                    'apmc': 'Padra_APMC'
                })
            t = nx.shortest_path_length(G, v_node, padra_node, weight=time_weight)
            dist_km = t / 3600 * 50
            direct_trips.append({
                'village': v['name'],
                'supply_kg': v['supply'] * 1000,
                'distance_km': dist_km
            })
            print(f"  {v['name']} → Padra_APMC: {t/60:.1f} minutes ({dist_km:.1f} km)")
        except nx.NetworkXNoPath:
            print(f"  ⚠️ No path for {v['name']}")

    # Metrics
    total_trips = len(first_level_trips) + len(second_level_trips) + len(direct_trips)
    total_distance = sum(t['distance_km'] for t in first_level_trips) + sum(t['distance_km'] for t in second_level_trips) + sum(t['distance_km'] for t in direct_trips)
    fixed_cost_per_cc = 5000
    total_fixed_cost = len(ccs) * fixed_cost_per_cc
    variable_cost = total_distance * 15
    total_cost = total_fixed_cost + variable_cost
    cost_per_tonne = total_cost / (villages['supply'].sum())
    total_cc_demand = villages[villages['cc'] != 'DIRECT']['supply'].sum()
    avg_load_per_trip = total_cc_demand / len(first_level_trips) if first_level_trips else 0
    vehicle_utilization = avg_load_per_trip / 15000 * 100 if first_level_trips else 0
    demand_fulfilled_pct = 100.0
    avg_travel_time_hours = total_distance / 50 / total_trips if total_trips > 0 else 0
    comp_time = time.time() - start_time

    results = {
        'scenario': scenario_name,
        'total_trips': total_trips,
        'total_distance_km': total_distance,
        'total_cost_rs': total_cost,
        'fixed_cost_rs': total_fixed_cost,
        'variable_cost_rs': variable_cost,
        'cost_per_tonne_rs': cost_per_tonne,
        'vehicle_utilization_pct': vehicle_utilization,
        'demand_fulfilled_pct': demand_fulfilled_pct,
        'avg_travel_time_hours': avg_travel_time_hours,
        'num_vehicles_used': len(first_level_trips) + len(second_level_trips) + len(direct_trips),
        'computation_time_sec': comp_time,
        'first_level_trips': first_level_trips,
        'second_level_trips': second_level_trips,
        'direct_trips': direct_trips,
        'route_geometries': route_geometries,
    }
    return results

# ------------------------------------------------------

summer_results = solve_scenario("Summer", time_weight='time_summer')
